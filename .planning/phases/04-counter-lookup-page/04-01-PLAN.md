```typescript
export async function getAllActiveStrategies(): Promise<CounterStrategyWithConditions[]> {
  if (!isSupabaseConfigured()) return [];

  const { data, error } = await supabase
    .from('counter_strategies')
    .select('*, strategy_conditions(*, characters(*)), team_compositions(*, team_members(*, characters(*)))')
    .eq('is_active', true)
    .order('priority');

  if (error) {
    if (import.meta.env.DEV) console.error('Error fetching all active strategies:', error);
    return [];
  }
  return (data ?? []) as CounterStrategyWithConditions[];
}
```

IMPORTANT: The key difference from existing `getCounterStrategies(templateId)` is:
- No `.eq('template_id', ...)` filter -- fetches ALL templates
- The `team_compositions` join is extended to `team_compositions(*, team_members(*, characters(*)))` to include team member character data for portrait rendering
